df %>%  #ctrl+shift+M
  select(where(is.numeric), -save_rate) %>% 
  pivot_longer(
    cols = everything(),
    names_to = 'variable',
    values_to = 'valor'
  ) %>%
  group_by(variable) %>% #ctrl+F - para cambiar el nombre de varias cosas al mismo tiempo
  summarise(n = length(valor),
            media = mean(valor),
            sd = sd(valor),
            mediana = median(valor),
            RIC = IQR(valor),
            Q1 = quantile(valor, 0.25),
            Q3 = quantile(valor, 0.75),
            minimo = min(valor),
            maximo = max(valor),
            asim = e1071::skewness(valor))
